let occupancy = {}; const DIRS = [ { name: 'UP', dc: 0, dr: -1 }, { name: 'DOWN', dc: 0, dr: 1 }, { name: 'LEFT', dc: -1, dr: 0 }, { name: 'RIGHT', dc: 1, dr: 0 } ]; // Corredor de Saída: Células a partir da cabeça até a borda function getCorridor(headC, headR, dir, occ) { let corridor = []; let c = headC + dir.dc; let r = headR + dir.dr; while (c >= 0 && c < COLS && r >= 0 && r < ROWS) { corridor.push(`${c},${r}`); c += dir.dc; r += dir.dr; } return corridor; } // Phase 2 - Backwards Builder (Python to JS) function generateLevel() { let placed = 0; let attempts = 0; let targetArrows = 8; while (placed < targetArrows && attempts < 800) { attempts++; let headC = Math.floor(Math.random() * COLS); let headR = Math.floor(Math.random() * ROWS); let dir = DIRS[Math.floor(Math.random() * DIRS.length)]; // Comprimento 2 a 4, traçado reto (Tangle 0.0) let length = Math.floor(Math.random() * 3) + 2; let body = []; let valid = true; for (let i = 0; i < length; i++) { // Corpo cresce na direção oposta à saída let bc = headC - (dir.dc * i); for (let cell of body) { if (aCorridor.includes(`${cell.c},${cell.r}`)) { blocksAny = true; break; } } if (blocksAny) break; } if (!blocksAny) continue; } // Registra a seta e bloqueia as células let arrowId = `arr_${placed}`; body.reverse(); // Armazena da Cauda -> Cabeça arrows.push({ id: arrowId, cells: body, head: {c: headC, r: headR}, dir: dir, exited: false, animOffset: 0, shake: 0 }); body.forEach(cell => occupancy[`${cell.c},${cell.r}`] = arrowId); placed++; } } generateLevel(); let animId; function draw() { if (GAME.currentLevel !== 11) return; ctx.clearRect(0, 0, canvas.width, canvas.height); // Grid pontilhado base ctx.fillStyle = 'rgba(255,255,255,0.03)'; else ctx.lineTo(cx, cy); } ctx.stroke(); // Desenho da Cabeça (Seta Direcional) let headCx = arr.head.c * cellW + cellW/2; let headCy = arr.head.r * cellH + cellH/2; ctx.translate(headCx, headCy); let angle = 0; const scaleX = canvas.width / rect.width; const scaleY = canvas.height / rect.height; const mx = (e.clientX - rect.left) * scaleX; const my = (e.clientY - rect.top) * scaleY; const c = Math.floor(mx / cellW); draw(); }
